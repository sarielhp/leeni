% Some preambles replace LaTeX's \@input with the primitive \input, so
% \begin{document} and every \include stop with "File `x.aux' not found" on a
% fresh build unless the .aux exists before the pass reads it.

\makeatletter
\let\@input\input
\makeatother
\begin{document}

Section~\ref{sec:intro} builds on a fresh checkout; so does Section~\ref{sec:chapter}.
\include{chapter}
\end{document}
